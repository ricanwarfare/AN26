// Minimal cfapi.h stub for mingw-w64 cross-compilation
// Cloud Files API declarations needed by cloudfiles.cpp
#pragma once

#include <windows.h>

#ifdef __cplusplus
extern "C" {
#endif

// Connection key
typedef LARGE_INTEGER CF_CONNECTION_KEY;

// Sync root registration flags
typedef enum {
    CF_REGISTER_FLAG_NONE = 0x00000000,
    CF_REGISTER_FLAG_UPDATE = 0x00000001,
    CF_REGISTER_FLAG_DISABLE_ON_DEMAND_POPULATION_ON_ROOT = 0x00000002,
    CF_REGISTER_FLAG_MARK_IN_SYNC_ON_ROOT = 0x00000004
} CF_REGISTER_FLAGS;

// Connect flags
typedef enum {
    CF_CONNECT_FLAG_NONE = 0x00000000,
    CF_CONNECT_FLAG_REQUIRE_PROCESS_INFO = 0x00000002,
    CF_CONNECT_FLAG_REQUIRE_FULL_FILE_PATH = 0x00000004
} CF_CONNECT_FLAGS;

// Hydration policy
typedef struct {
    USHORT Primary;
    USHORT Modifier;
} CF_HYDRATION_POLICY;

#define CF_HYDRATION_POLICY_FULL 3

// Population policy
typedef struct {
    USHORT Primary;
    USHORT Modifier;
} CF_POPULATION_POLICY;

#define CF_POPULATION_POLICY_FULL 3

// InSync policy
typedef ULONG CF_INSYNC_POLICY;
#define CF_INSYNC_POLICY_TRACK_ALL 0xFFFFFFFF

// HardLink policy
typedef ULONG CF_HARDLINK_POLICY;
#define CF_HARDLINK_POLICY_NONE 0

// PlaceholderManagement policy
typedef ULONG CF_PLACEHOLDER_MANAGEMENT_POLICY;
#define CF_PLACEHOLDER_MANAGEMENT_POLICY_DEFAULT 0

// Sync policies
typedef struct {
    ULONG StructSize;
    CF_HYDRATION_POLICY Hydration;
    CF_POPULATION_POLICY Population;
    CF_INSYNC_POLICY InSync;
    CF_HARDLINK_POLICY HardLink;
    CF_PLACEHOLDER_MANAGEMENT_POLICY PlaceholderManagement;
} CF_SYNC_POLICIES;

// Sync registration — layout must match the real Windows SDK definition
// or CfRegisterSyncRoot returns E_INVALIDARG (0x80070057).
typedef struct {
    ULONG    StructSize;
    LPCWSTR  ProviderName;
    LPCWSTR  ProviderVersion;
    LPCVOID  SyncRootIdentity;          // required, non-NULL
    ULONG    SyncRootIdentityLength;
    LPCVOID  FileIdentity;              // required, non-NULL
    ULONG    FileIdentityLength;
    GUID     ProviderId;
} CF_SYNC_REGISTRATION;

// Process info
typedef struct {
    ULONG StructSize;
    ULONG_PTR ProcessId;
    LPCWSTR ImagePath;
    LPCWSTR PackageName;
    LPCWSTR ApplicationId;
    LPCWSTR CommandLine;
    ULONG SessionId;
} CF_PROCESS_INFO;

// Callback info
typedef struct {
    ULONG StructSize;
    CF_CONNECTION_KEY ConnectionKey;
    PVOID CallbackContext;
    LPCWSTR VolumeGuidName;
    LPCWSTR VolumeDosName;
    ULONG VolumeSerialNumber;
    LARGE_INTEGER SyncRootFileId;
    LARGE_INTEGER SyncRootIdentity;
    ULONG SyncRootIdentityLength;
    LARGE_INTEGER FileId;
    LARGE_INTEGER FileSize;
    LPCWSTR NormalizedPath;
    LARGE_INTEGER TransferKey;
    UCHAR PriorityHint;
    ULONG CorrelationVector;
    CF_PROCESS_INFO* ProcessInfo;
    LARGE_INTEGER RequestKey;
} CF_CALLBACK_INFO;

// Callback types
typedef enum {
    CF_CALLBACK_TYPE_FETCH_DATA = 0,
    CF_CALLBACK_TYPE_VALIDATE_DATA = 1,
    CF_CALLBACK_TYPE_CANCEL_FETCH_DATA = 2,
    CF_CALLBACK_TYPE_FETCH_PLACEHOLDERS = 3,
    CF_CALLBACK_TYPE_CANCEL_FETCH_PLACEHOLDERS = 4,
    CF_CALLBACK_TYPE_OPEN_COMPLETION = 5,
    CF_CALLBACK_TYPE_CLOSE_COMPLETION = 6,
    CF_CALLBACK_TYPE_DEHYDRATE = 7,
    CF_CALLBACK_TYPE_DEHYDRATE_COMPLETION = 8,
    CF_CALLBACK_TYPE_DELETE = 9,
    CF_CALLBACK_TYPE_DELETE_COMPLETION = 10,
    CF_CALLBACK_TYPE_RENAME = 11,
    CF_CALLBACK_TYPE_RENAME_COMPLETION = 12,
    CF_CALLBACK_TYPE_NONE = 0xFFFF
} CF_CALLBACK_TYPE;

// Callback parameters (union of all callback params)
typedef struct {
    ULONG ParamSize;
    union {
        struct {
            NTSTATUS CompletionStatus;
            LARGE_INTEGER PlaceholderTotalCount;
            ULONG EntriesProcessed;
            PVOID PlaceholderArray;
        } TransferPlaceholders;
        struct {
            NTSTATUS CompletionStatus;
            LARGE_INTEGER Length;
            LARGE_INTEGER Offset;
            PVOID Buffer;
        } TransferData;
    };
} CF_OPERATION_PARAMETERS;

// Operation types
typedef enum {
    CF_OPERATION_TYPE_TRANSFER_DATA = 0,
    CF_OPERATION_TYPE_RETRIEVE_DATA = 1,
    CF_OPERATION_TYPE_ACK_DATA = 2,
    CF_OPERATION_TYPE_RESTART_HYDRATION = 3,
    CF_OPERATION_TYPE_TRANSFER_PLACEHOLDERS = 4,
    CF_OPERATION_TYPE_ACK_DEHYDRATE = 5,
    CF_OPERATION_TYPE_ACK_DELETE = 6,
    CF_OPERATION_TYPE_ACK_RENAME = 7
} CF_OPERATION_TYPE;

typedef struct {
    ULONG StructSize;
    CF_OPERATION_TYPE Type;
    CF_CONNECTION_KEY ConnectionKey;
    LARGE_INTEGER TransferKey;
    ULONG CorrelationVector;
    LARGE_INTEGER RequestKey;
} CF_OPERATION_INFO;

#define CF_SIZE_OF_OP_PARAM(field) (FIELD_OFFSET(CF_OPERATION_PARAMETERS, field) + sizeof(((CF_OPERATION_PARAMETERS*)0)->field))

// Forward declare for callback typedef
typedef struct _CF_CALLBACK_PARAMETERS CF_CALLBACK_PARAMETERS;

// Callback function pointer
typedef void (CALLBACK *CF_CALLBACK)(
    const CF_CALLBACK_INFO* CallbackInfo,
    const CF_CALLBACK_PARAMETERS* CallbackParameters
);

// Actual definition
struct _CF_CALLBACK_PARAMETERS {
    ULONG ParamSize;
};

// Callback registration entry
typedef struct {
    CF_CALLBACK_TYPE Type;
    CF_CALLBACK Callback;
} CF_CALLBACK_REGISTRATION;

// API functions — loaded dynamically at runtime
typedef HRESULT (WINAPI *fnCfRegisterSyncRoot)(
    LPCWSTR SyncRootPath,
    const CF_SYNC_REGISTRATION* Registration,
    const CF_SYNC_POLICIES* Policies,
    CF_REGISTER_FLAGS RegisterFlags
);

typedef HRESULT (WINAPI *fnCfConnectSyncRoot)(
    LPCWSTR SyncRootPath,
    const CF_CALLBACK_REGISTRATION* CallbackTable,
    PVOID CallbackContext,
    CF_CONNECT_FLAGS ConnectFlags,
    CF_CONNECTION_KEY* ConnectionKey
);

typedef HRESULT (WINAPI *fnCfDisconnectSyncRoot)(CF_CONNECTION_KEY ConnectionKey);
typedef HRESULT (WINAPI *fnCfUnregisterSyncRoot)(LPCWSTR SyncRootPath);
typedef HRESULT (WINAPI *fnCfExecute)(const CF_OPERATION_INFO* OpInfo, CF_OPERATION_PARAMETERS* OpParams);

#ifdef __cplusplus
}
#endif
